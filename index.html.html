<script>
    const tg = window.Telegram.WebApp;
    const sfxSpin = new Audio('https://www.soundjay.com/buttons/sounds/button-20.mp3');
    const sfxThunder = new Audio('https://www.soundjay.com/nature/sounds/thunder-1.mp3');
    const sfxWin = new Audio('https://www.myinstants.com/media/sounds/jackpot.mp3');

    function initAudio() { sfxSpin.play(); sfxSpin.pause(); }

    let balance = 10000;
    let bet = 300;
    const items = ['👑', '💎', '⏳', '💍', '⚡', '🔱', '🟡', '🔴'];

    function handleSpin() {
        if (balance < bet) return;
        balance -= bet;
        document.getElementById('balance').innerText = balance.toLocaleString();
       
        // إرسال للخصم للبوت
        tg.sendData(JSON.stringify({ "action": "spin", "amount": bet }));

        sfxSpin.play();
        const board = document.getElementById('board');
        board.innerHTML = '';

        for (let i = 0; i < 12; i++) {
            const div = document.createElement('div');
            div.className = 'symbol spin-anim';
            div.innerText = items[Math.floor(Math.random() * items.symbols.length)]; // تعديل صغير
            board.appendChild(div);
           
            setTimeout(() => {
                div.classList.remove('spin-anim');
                div.innerText = items[Math.floor(Math.random() * items.length)];
            }, i * 50);
        }

        setTimeout(checkResult, 1000);
    }

    function checkResult() {
        if (Math.random() > 0.8) {
            sfxThunder.play();
            document.getElementById('flash').style.animation = 'flash 0.5s';
            setTimeout(() => {
                sfxWin.play();
                alert("صعقة زيوس! ربحت " + (bet * 5));
                balance += (bet * 5);
                document.getElementById('balance').innerText = balance.toLocaleString();
                tg.sendData(JSON.stringify({ "action": "win", "amount": bet * 5 }));
                document.getElementById('flash').style.animation = '';
            }, 500);
        }
    }
</script>
